**1. Exponential backoff helper `def fetch_with_backoff(url, headers, body, max_retries=3, base_delay=1.0) -> dict`**

```python
import time

def fetch_with_backoff(url: str, headers: dict, body: dict,
                       max_retries: int = 3, base_delay: float = 1.0) -> dict:
    """POST request with exponential backoff on non-2xx responses."""
    for attempt in range(max_retries):
        response = requests.post(url, headers=headers, json=body)
        if response.status_code == 200:
            return response.json()
        wait = base_delay * (2 ** attempt)
        print(f"[backoff] HTTP {response.status_code} — retrying in {wait:.1f}s "
              f"(attempt {attempt + 1}/{max_retries})", file=sys.stderr)
        time.sleep(wait)
    # Final attempt — raise to let caller handle
    response.raise_for_status()
```

**2. Paginated `search_places()` — replace the existing single-page version**

```python
PLACES_URL = "https://places.googleapis.com/v1/places:searchText"

def search_places(query: str, city: str, max_results: int, api_key: str) -> list[dict]:
    headers = {
        "Content-Type": "application/json",
        "X-Goog-Api-Key": api_key,
        "X-Goog-FieldMask": (
            "places.displayName,places.formattedAddress,"
            "places.internationalPhoneNumber,places.websiteUri,"
            "places.businessStatus,nextPageToken"
        ),
    }
    body: dict = {"textQuery": f"{query} em {city}", "languageCode": "pt-BR"}
    all_places: list[dict] = []

    while len(all_places) < max_results:
        data = fetch_with_backoff(PLACES_URL, headers, body)
        places = data.get("places", [])
        all_places.extend(places)
        next_token = data.get("nextPageToken")
        if not next_token or not places:
            break
        body = {"textQuery": f"{query} em {city}", "languageCode": "pt-BR",
                "pageToken": next_token}

    return all_places[:max_results]
```

**`def save_lead(client: Client, lead: dict, index: int, total: int) -> str`**

```python
def save_lead(client: Client, lead: dict, index: int, total: int) -> str:
    """Upsert lead to Supabase. Returns 'novo', 'atualizado', or raises on error."""
    # Fetch existing to detect insert vs update
    existing = (
        client.table("leads")
        .select("id")
        .eq("name", lead["name"])
        .eq("city", lead["city"])
        .execute()
    )
    is_update = len(existing.data) > 0

    client.table("leads").upsert(lead, on_conflict="name,city").execute()

    status = "atualizado" if is_update else "salvo"
    print(f"[{index}/{total}] {lead['name']} — {status}")
    return status
```

    total = len(leads)

    # Save with progress bar
    new_count = 0
    updated_count = 0
    error_count = 0

    with tqdm(total=total, desc="Processando", unit="lead") as pbar:
        for i, lead in enumerate(leads, start=1):
            try:
                status = save_lead(client, lead, i, total)
                if status == "salvo":
                    new_count += 1
                else:
                    updated_count += 1
            except Exception as e:
                error_count += 1
                print(f"[{i}/{total}] {lead.get('name', '?')} — erro: {e}", file=sys.stderr)
            finally:
                pbar.update(1)

    print(f"\nConcluído: {new_count} novos, {updated_count} atualizados, {error_count} erros")
